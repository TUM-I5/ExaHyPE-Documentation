Part 2 - Solution to Exercise 1

Initial conditions: 

void tutorials::exahype2::swe::ADERDGSolver::initialCondition(
  double* const NOALIAS                   Q,
  const tarch::la::Vector<DIMENSIONS, double>& cellCentre,
  const tarch::la::Vector<DIMENSIONS, double>& cellSize
) {
  Q[Shortcuts::hu + 0] = 0.0;
  Q[Shortcuts::hu + 1] = 0.0;
  Q[Shortcuts::h]      = cellCentre[0] < 0.0 ? 1.0 : 2.0;
  Q[Shortcuts::b]      = 0.0;
}

Boundary conditions: 

void tutorials::exahype2::swe::ADERDGSolver::boundaryConditions(
  const double* const NOALIAS             Qinside,
  double* const NOALIAS                   Qoutside,
  const tarch::la::Vector<DIMENSIONS, double>& faceCentre,
  const tarch::la::Vector<DIMENSIONS, double>& cellSize,
  const double                                 time,
  const int                                    normal
) {
  Qoutside[Shortcuts::h]      = 1.0;
  Qoutside[Shortcuts::hu + 0] = 0.0;
  Qoutside[Shortcuts::hu + 1] = 0.0;
  Qoutside[Shortcuts::b]      = 0.0;
}

Max eigenvalue: 

double tutorials::exahype2::swe::ADERDGSolver::maxEigenvalue(
  const double* const NOALIAS             Q,
  const tarch::la::Vector<DIMENSIONS, double>& cellCentre,
  const tarch::la::Vector<DIMENSIONS, double>& cellSize,
  const double                                 time,
  const double                                 timeStepSize,
  const int                                    normal
) {
  constexpr double g = 9.81;
  const double u = Q[Shortcuts::hu + normal] / Q[Shortcuts::h];
  const double c = std::sqrt(g * Q[Shortcuts::h]);
  return std::max(std::abs(u + c), std::abs(u - c));
}

Flux (with the pressure term still in the flux): 

void tutorials::exahype2::swe::ADERDGSolver::flux(
  const double* const NOALIAS             Q,
  const tarch::la::Vector<DIMENSIONS, double>& cellCentre,
  const tarch::la::Vector<DIMENSIONS, double>& cellSize,
  const double                                 time,
  const double                                 timeStepSize,
  const int                                    normal,
  double* const NOALIAS                   F
) {
  const double     ih = 1.0 / Q[Shortcuts::h];
  constexpr double g  = 9.81;
  F[Shortcuts::h]      = Q[Shortcuts::hu + normal];
  F[Shortcuts::hu + 0] = Q[Shortcuts::hu + normal] * Q[Shortcuts::hu + 0] * ih;
  F[Shortcuts::hu + 1] = Q[Shortcuts::hu + normal] * Q[Shortcuts::hu + 1] * ih;
  F[Shortcuts::b]      = 0.0;
  F[Shortcuts::hu + normal] += 0.5 * g * Q[Shortcuts::h] * Q[Shortcuts::h];
}
